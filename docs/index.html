Command Line

Full parametric control from the terminal

# Generate an interdigitated electrode
python electrode_generator.py --type ide --fingers 10 --finger-w 0.5 --gap 0.3 \
    --finger-l 15 --format both --output my_electrode

# Generate G-code for laser cutting
python electrode_generator.py --type spiral --turns 5 --format gcode \
    --feed-rate 800 --laser-power 500 --output spiral_lig

# Export layers separately for multi-material Cricut
python electrode_generator.py --type ide --layers ELECTRODE_A ELECTRODE_B \
    --format dxf --output my_ide_layers
